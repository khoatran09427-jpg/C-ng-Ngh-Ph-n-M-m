
FROM php:8.2-apache

LABEL description="Online vehicle rental system - PHP 8.2 + Apache"

WORKDIR /var/www/html

# cai dat PDO MySQL
RUN docker-php-ext-install pdo_mysql

# Sao chep ma nguon website
COPY src/ /var/www/html/

# Tao thu muc luu anh xe
RUN mkdir -p /var/www/html/uploads/vehicles \
    && chown -R www-data:www-data /var/www/html/uploads \
    && chmod -R 755 /var/www/html/uploads

EXPOSE 80

# kiem tra apache
HEALTHCHECK --interval=30s --timeout=5s --start-period=15s --retries=3 \
    CMD php -r '$s=@fsockopen("127.0.0.1",80); if ($s) {fclose($s); exit(0);} exit(1);'

CMD ["apache2-foreground"]
